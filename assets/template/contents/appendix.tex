  \begin{equation*}
    w_i=\frac{x_i}{\sum_{j=1}^{n}x_j},\qquad
    x_i\geq 0,\quad \sum_{j=1}^{n}x_j>0
  \end{equation*}
  \vspace{3mm}
  \begin{itemize}
    \setlength{\itemsep}{3mm}
    \item $x_i$：第 $i$ 项非负输入。
    \item $w_i$：归一化后的权重，满足 $\sum_i w_i=1$。
    \item 分母为正是该表达式的使用条件。
  \end{itemize}
